cmake_minimum_required(VERSION 3.22.1)
project(codua2a C)
set(CMAKE_C_STANDARD 11)
get_filename_component(CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../core" ABSOLUTE)
if(NOT EXISTS "${CORE}/3rd/libdeflate/lib/deflate_compress.c")
    message(FATAL_ERROR "Initialize core and its libdeflate submodule; see README.md")
endif()

# Adapt the executable entry point at build time without modifying the submodule.
file(READ "${CORE}/xlua/xnet_main.c" RUNNER)
set(HOOK "static void preload_modules(lua_State* L) {")
string(FIND "${RUNNER}" "${HOOK}" HOOK_POS)
if(HOOK_POS EQUAL -1)
    message(FATAL_ERROR "Upstream runner changed: review the Android bridge insertion")
endif()
string(REPLACE "${HOOK}" "${HOOK}\n    extern void codua2a_install(lua_State*);\n    codua2a_install(L);" RUNNER "${RUNNER}")
string(REPLACE "int main(int argc, char** argv)" "int codua2a_run(int argc, char** argv)" RUNNER "${RUNNER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/codua2a_runner.c" "${RUNNER}")
set(SOURCES)
foreach(SRC xargs xpoll xsock xchannel xthread xtimer xlog xshared xrecord xframe_aead)
    list(APPEND SOURCES "${CORE}/${SRC}.c")
endforeach()
foreach(MOD xthread xnet xnet_tls cmsgpack xtimer xcompress xshared xrecord xutils)
    list(APPEND SOURCES "${CORE}/xlua/lua_${MOD}.c")
endforeach()
file(GLOB TLS_SOURCES CONFIGURE_DEPENDS "${CORE}/3rd/mbedtls3/library/*.c")
file(GLOB DEFLATE_SOURCES CONFIGURE_DEPENDS "${CORE}/3rd/libdeflate/lib/*.c")
if(ANDROID_ABI STREQUAL "arm64-v8a")
    list(APPEND DEFLATE_SOURCES "${CORE}/3rd/libdeflate/lib/arm/cpu_features.c")
else()
    list(APPEND DEFLATE_SOURCES "${CORE}/3rd/libdeflate/lib/x86/cpu_features.c")
endif()
add_library(codua2a SHARED bridge.c android_daemon.c
    "${CMAKE_CURRENT_BINARY_DIR}/codua2a_runner.c" ${SOURCES}
    "${CORE}/3rd/yyjson.c" ${TLS_SOURCES} ${DEFLATE_SOURCES})
target_include_directories(codua2a PRIVATE "${CORE}" "${CORE}/3rd/libdeflate" "${CORE}/3rd/mbedtls3/include")
target_compile_definitions(codua2a PRIVATE LUA_EMBEDDED XNET_WITH_HTTP=1 XNET_WITH_HTTPS=1
    XNET_WITH_XDEBUG=0 XNET_WITH_XPROC=0 XMACRO_USE_RPMALLOC=0)
# Also supports 16 KiB page-size devices with this pinned NDK.
target_link_options(codua2a PRIVATE "-Wl,-z,max-page-size=16384")
target_link_libraries(codua2a log android m dl)
